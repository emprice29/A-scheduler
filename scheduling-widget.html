<script>

  // ---- CONFIG: set this after you deploy the Apps Script Web App ----

  const WEB_APP_URL = https://script.google.com/macros/s/AKfycbxC4gQgxZdC8cJYHRjyvQpyYAa3oEG_F6ESrRUKnzmp8OEVx7Q3rOyYQEN0dv9I45SwHQ/exec;


  const state = {

    selected: [] // { date, slotKey, dayLabel, slotLabel }

  };


  const panels = {

    1: document.getElementById('panel-1'),

    2: document.getElementById('panel-2'),

    3: document.getElementById('panel-3')

  };

  const dots = document.querySelectorAll('.step-dot');


  function goToStep(n) {

    Object.keys(panels).forEach(k => panels[k].classList.toggle('active', Number(k) === n));

    dots.forEach((d, i) => {

      d.classList.toggle('active', i === n - 1);

      d.classList.toggle('done', i < n - 1);

    });

    window.scrollTo({ top: 0, behavior: 'smooth' });

  }


  document.getElementById('to-step-2').addEventListener('click', () => {

    const required = ['f-name', 'f-phone', 'f-email', 'f-address', 'f-city', 'f-sqft'];

    for (const id of required) {

      const el = document.getElementById(id);

      if (!el.value.trim()) { el.focus(); return; }

    }

    goToStep(2);

    if (!state.loaded) loadAvailability();

  });


  document.getElementById('back-to-1').addEventListener('click', () => goToStep(1));


  function formatDayLabel(dateStr) {

    const d = new Date(dateStr + 'T00:00:00');

    return d.toLocaleDateString(undefined, { weekday: 'long', month: 'short', day: 'numeric' });

  }


  function loadAvailability() {

    state.loaded = true;

    const loadingEl = document.getElementById('avail-loading');

    const errorEl = document.getElementById('avail-error');

    const daysEl = document.getElementById('avail-days');

    loadingEl.style.display = 'block';

    errorEl.style.display = 'none';


    fetch(WEB_APP_URL + '?action=availability')

      .then(r => r.json())

      .then(data => {

        loadingEl.style.display = 'none';

        if (data.error) throw new Error(data.error);

        daysEl.innerHTML = '';

        data.days.forEach(day => {

          const anyOpen = day.slots.some(s => s.available);

          if (!anyOpen) return; // hide fully booked days to keep the list short

          const block = document.createElement('div');

          block.className = 'day-block';

          block.innerHTML =

            '<div class="day-name">' + formatDayLabel(day.date) +

            '<span class="sub">' + day.date + '</span></div>' +

            '<div class="slot-row"></div>';

          const row = block.querySelector('.slot-row');

          day.slots.forEach(slot => {

            const btn = document.createElement('button');

            btn.type = 'button';

            btn.className = 'slot-btn';

            btn.textContent = slot.label;

            btn.disabled = !slot.available;

            btn.addEventListener('click', () => toggleSlot(day.date, slot, btn));

            row.appendChild(btn);

          });

          daysEl.appendChild(block);

        });

        if (!daysEl.children.length) {

          errorEl.textContent = 'No open times in the next few weeks — please call or text us directly.';

          errorEl.style.display = 'block';

        }

      })

      .catch(err => {

        loadingEl.style.display = 'none';

        errorEl.textContent = 'Could not load open times right now. Please call or text us, or try again shortly.';

        errorEl.style.display = 'block';

      });

  }


  function toggleSlot(date, slot, btn) {

    const idx = state.selected.findIndex(s => s.date === date && s.slotKey === slot.key);

    if (idx > -1) {

      state.selected.splice(idx, 1);

      btn.classList.remove('selected');

    } else {

      state.selected.push({ date, slotKey: slot.key, dayLabel: formatDayLabel(date), slotLabel: slot.label });

      btn.classList.add('selected');

    }

    updateSummary();

  }


  function updateSummary() {

    const el = document.getElementById('selected-summary');

    const submitBtn = document.getElementById('to-step-3');

    submitBtn.disabled = state.selected.length === 0;

    el.textContent = state.selected.length

      ? state.selected.length + ' time' + (state.selected.length > 1 ? 's' : '') + ' selected'

      : '';

  }


  document.getElementById('to-step-3').addEventListener('click', () => {

    const submitBtn = document.getElementById('to-step-3');

    submitBtn.disabled = true;

    submitBtn.textContent = 'Submitting…';


    const payload = {

      name: document.getElementById('f-name').value.trim(),

      phone: document.getElementById('f-phone').value.trim(),

      email: document.getElementById('f-email').value.trim(),

      realtorName: document.getElementById('f-realtor-name').value.trim(),

      realtorEmail: document.getElementById('f-realtor-email').value.trim(),

      address: document.getElementById('f-address').value.trim(),

      city: document.getElementById('f-city').value.trim(),

      sqft: document.getElementById('f-sqft').value.trim(),

      services: Array.from(document.querySelectorAll('.svc:checked')).map(el => el.value),

      selectedSlots: state.selected.map(s => ({ date: s.date, slotKey: s.slotKey }))

    };


    fetch(WEB_APP_URL, {

      method: 'POST',

      headers: { 'Content-Type': 'text/plain;charset=utf-8' }, // avoids CORS preflight against Apps Script

      body: JSON.stringify(payload)

    })

      .then(r => r.json())

      .then(data => {

        if (data.error) throw new Error(data.error);

        const recap = document.getElementById('recap');

        recap.innerHTML = state.selected

          .map(s => '<div class="recap-item">' + s.dayLabel + ' — ' + s.slotLabel + '</div>')

          .join('');

        goToStep(3);

      })

      .catch(err => {

        alert('Something went wrong submitting your request. Please call or text us directly.');

        submitBtn.disabled = false;

        submitBtn.textContent = 'Submit request';

      });

  });

</script>
